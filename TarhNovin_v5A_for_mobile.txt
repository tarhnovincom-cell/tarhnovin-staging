def _validate_environment():
 if os.environ.get('APP_ENV') != 'staging':raise SystemExit('APP_ENV must be staging; demo data only')
 if len(os.environ.get('SESSION_SECRET',''))<32:raise SystemExit('Missing SESSION_SECRET')
 if len(os.environ.get('STAGING_GATE_PASSWORD',''))<16:raise SystemExit('Missing STAGING_GATE_PASSWORD')
 if len(os.environ.get('ADMIN_PASSWORD',''))<12:raise SystemExit('Missing ADMIN_PASSWORD')
 os.environ.setdefault('STAGING_GATE_USERNAME','review')
 os.environ.setdefault('DATABASE_URL','sqlite:///./data/staging-preview.sqlite3')
def _unpack():
 root=Path(__file__).resolve().parent/'.tarhnovin_staging_runtime'
 root.mkdir(parents=True,exist_ok=True)
 binary=base64.b85decode(''.join(_ARCHIVE))
 with tarfile.open(fileobj=io.BytesIO(binary),mode='r:gz') as t:
  for member in t.getmembers():
   destination=(root/member.name).resolve()
   if not destination.is_relative_to(root.resolve()) or not member.isfile():raise SystemExit('Invalid archive entry')
   destination.parent.mkdir(parents=True,exist_ok=True)
   with t.extractfile(member) as source,open(destination,'wb') as output:
    while True:
     b=source.read(65536)
     if not b:break
     output.write(b)
 return root
